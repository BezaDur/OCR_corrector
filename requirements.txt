# Runtime dependencies of the parent pipeline and the child detector.
#
#   pip install -r requirements.txt
#
# Grouped by what needs them:
#   - detector / corrector:  numpy, joblib, lightgbm, scikit-learn, symspellpy
#   - vendored morphology:    pydantic  (vendor/segmenter/morph_types.py)
#   - training + evaluation:  pandas
#   - page synthesis / OCR:   opencv-python, Pillow, fonttools
#   - KenLM:  not on PyPI. Install "kenlm" itself (pip install kenlm, or the
#     https://github.com/kpu/kenlm build) and supply a model file; the corrector
#     needs a word 5-gram, the detector's char-LM column is optional.
numpy
joblib
lightgbm
scikit-learn
symspellpy==6.10.0

# corrector then runs with morphology silently disabled (lambda_morph cancels
# out). It was missing from this file, which is why the failure looked like a
# missing module rather than a missing dependency.
pydantic
pandas
opencv-python
Pillow
fonttools
